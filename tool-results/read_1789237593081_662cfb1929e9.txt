    65→// ─── Aliphia CSV Parser ───────────────────────────────────────────
    66→/**
    67→
    68→- Aliphia exports CSV with these common columns (Arabic + English headers):
    69→- Invoice No, Date, Customer Name, Customer Phone, Customer Address,
    70→- Item Name, Item Description, Qty, Unit Price, Total, Shipping, Paid, Notes, Due Date
    71→- The parser is flexible — tries multiple header variants.
    72→  */
    73→  function parseAliphiaCSV(text) {
    74→  const lines = text.trim().split(/\r?\n/);
    75→  if (lines.length < 2) return { invoices: [], errors: ["الملف فارغ أو غير صحيح"] };
    76→
    77→// Parse header row — normalize
    78→const rawHeaders = lines[0].split(",").map(h => h.trim().replace(/^"|"$/g, "").toLowerCase());
    79→
    80→// Header mapping: aliphia field → our field
    81→const colMap = {
    82→invNum:         ["invoice no","invoice number","رقم الفاتورة","inv no","رقم"],
    83→date:           ["date","تاريخ","invoice date","تاريخ الفاتورة"],
    84→dueDate:        ["due date","تاريخ الاستحقاق","expiry date"],
    85→clientName:     ["customer name","client name","اسم العميل","الاسم","name","customer"],
    86→clientPhone:    ["phone","telephone","mobile","هاتف","رقم الهاتف","customer phone","client phone","جوال"],
    87→clientAddress:  ["address","عنوان","customer address","client address","العنوان"],
    88→itemName:       ["item name","product name","item","product","المنتج","اسم المنتج","الصنف"],
    89→itemDesc:       ["description","item description","وصف","الوصف","desc"],
    90→qty:            ["qty","quantity","كمية","الكمية"],
    91→price:          ["unit price","price","سعر","سعر الوحدة","السعر"],
    92→shipping:       ["shipping","delivery","توصيل","شحن"],
    93→paid:           ["paid","مدفوع","paid amount","المدفوع"],
    94→notes:          ["notes","ملاحظات","note","remarks"],
    95→};
    96→
    97→// Find column index for each field
    98→const idx = {};
    99→for (const [field, variants] of Object.entries(colMap)) {
   100→for (const v of variants) {
   101→const i = rawHeaders.findIndex(h => h.includes(v));
   102→if (i !== -1) { idx[field] = i; break; }
   103→}
   104→if (idx[field] === undefined) idx[field] = -1;
   105→}
   106→
   107→// Parse data rows
   108→const rows = lines.slice(1).map(line => {
   109→// Handle quoted CSV values
   110→const cols = [];
   111→let cur = "", inQ = false;
   112→for (let i = 0; i < line.length; i++) {
   113→const ch = line[i];
   114→if (ch === '"') { inQ = !inQ; }
   115→else if (ch === ',' && !inQ) { cols.push(cur.trim()); cur = ""; }
   116→else cur += ch;
   117→}
   118→cols.push(cur.trim());
   119→return cols;
   120→}).filter(r => r.some(c => c.length > 0));
   121→
   122→const get = (row, field) => {
   123→const i = idx[field];
   124→return i >= 0 && i < row.length ? row[i].replace(/^"|"$/g, "").trim() : "";
   125→};
   126→
   127→// Group rows by invoice number (Aliphia may repeat invoice rows for multiple items)
   128→const invMap = {};
   129→const errors = [];
   130→
   131→rows.forEach((row, ri) => {
   132→if (!row.length || row.every(c => !c)) return;
   133→
   134→let invNum = get(row, "invNum") || `IMP${ri+1}`;
   135→const key = invNum;
   136→
   137→// Normalize date: handle dd/mm/yyyy → yyyy-mm-dd
   138→const parseDate = s => {
   139→  if (!s) return today();
   140→  // Already yyyy-mm-dd
   141→  if (/^\d{4}-\d{2}-\d{2}/.test(s)) return s.substring(0, 10);
   142→  // dd/mm/yyyy or dd-mm-yyyy
   143→  const m = s.match(/^(\d{1,2})[\/\-](\d{1,2})[\/\-](\d{4})/);
   144→  if (m) return `${m[3]}-${m[2].padStart(2,"0")}-${m[1].padStart(2,"0")}`;
   145→  return today();
   146→};
   147→
   148→const itemName = get(row, "itemName");
   149→const itemDesc = get(row, "itemDesc");
   150→const qty = pN(get(row, "qty")) || 1;
   151→const price = pN(get(row, "price"));
   152→const item = { name: itemName || "منتج", desc: itemDesc, qty, price };
   153→
   154→if (invMap[key]) {
   155→  // Add item to existing invoice
   156→  if (itemName || price) invMap[key].items.push(item);
   157→} else {
   158→  invMap[key] = {
   159→    invNum,
   160→    clientName: get(row, "clientName") || "عميل",
   161→    clientPhone: norm(get(row, "clientPhone")),
   162→    clientAddress: get(row, "clientAddress"),
   163→    items: (itemName || price) ? [item] : [{ name: "منتج", desc: "", qty: 1, price: 0 }],
   164→    shipping: pN(get(row, "shipping")),
   165→    date: parseDate(get(row, "date")),
   166→    dueDate: parseDate(get(row, "dueDate")) || addD(today(), 30),
   167→    paid: pN(get(row, "paid")),
   168→    notes: get(row, "notes"),
   169→  };
   170→}
   171→
   172→});
   173→
   174→const invoices = Object.values(invMap);
   175→return { invoices, errors, detectedCols: rawHeaders };
   176→}
