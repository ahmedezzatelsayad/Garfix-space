     1→"use client";
     2→
     3→import { useState, useEffect, useCallback, useRef } from "react";
     4→import { api } from "../api";
     5→
     6→const pN = v => parseFloat(String(v ?? 0).replace(/[^\d.]/g, "")) || 0;
     7→const today = () => new Date().toISOString().split("T")[0];
     8→
     9→const CONF_COLOR = c =>
    10→  c >= 0.85 ? "#16a34a" : c >= 0.65 ? "#d97706" : "#dc2626";
    11→const CONF_LABEL = c =>
    12→  c >= 0.85 ? "عالية" : c >= 0.65 ? "متوسطة" : "منخفضة";
    13→const CONF_BG = c =>
    14→  c >= 0.85 ? "#dcfce7" : c >= 0.65 ? "#fef3c7" : "#fee2e2";
    15→
    16→export default function AIBulkProcessor({ company, onPurchaseSaved }) {
    17→  const col = company?.color || "#1e3a5f";
    18→  const [step, setStep] = useState(0); // 0=input, 1=processing, 2=review, 3=catalog, 4=done
    19→  const [rawText, setRawText] = useState("");
    20→  const [aiItems, setAiItems] = useState([]);
    21→  const [error, setError] = useState("");
    22→  const [catalog, setCatalog] = useState([]);
    23→  const [catalogLoading, setCatalogLoading] = useState(false);
    24→  const [toast, setToast] = useState(null);
    25→  const [purchaseForm, setPurchaseForm] = useState({ supplier: "", date: today(), notes: "" });
    26→  const [saving, setSaving] = useState(false);
    27→  const [newProduct, setNewProduct] = useState({ name: "", aliases: "", purchasePrice: "", sellingPrice: "" });
    28→  const [addingProduct, setAddingProduct] = useState(false);
    29→  const [editingCatalogId, setEditingCatalogId] = useState(null);
    30→  const textRef = useRef(null);
    31→
    32→  const toast_ = msg => { setToast(msg); setTimeout(() => setToast(null), 2800); };
    33→
    34→  const loadCatalog = useCallback(async () => {
    35→    setCatalogLoading(true);
    36→    try {
    37→      const data = await api.getCatalog(company?.sk);
    38→      setCatalog(data || []);
    39→    } catch { setCatalog([]); }
    40→    finally { setCatalogLoading(false); }
    41→  }, [company]);
    42→
    43→  useEffect(() => { loadCatalog(); }, [loadCatalog]);
    44→
    45→  // ── AI Processing ─────────────────────────────────────────────────
    46→  const processWithAI = async () => {
    47→    if (!rawText.trim()) return;
    48→    setStep(1);
    49→    setError("");
    50→    try {
    51→      const res = await fetch("/api/ai/process-items", {
    52→        method: "POST",
    53→        headers: { "Content-Type": "application/json" },
    54→        body: JSON.stringify({ rawText: rawText.trim(), catalog }),
    55→      });
    56→      if (!res.ok) {
    57→        const err = await res.json().catch(() => ({}));
    58→        throw new Error(err.error || `HTTP ${res.status}`);
    59→      }
    60→      const data = await res.json();
    61→      const items = (data.items || []).map((it, i) => ({
    62→        ...it,
    63→        _id: i,
    64→        quantity: Number(it.quantity) || 1,
    65→        confidence: Number(it.confidence) || 0.5,
    66→        purchasePrice: it.purchasePrice ?? (
    67→          catalog.find(c => c.id === it.catalogId)?.purchasePrice ?? ""
    68→        ),
    69→        edited: false,
    70→      }));
    71→      setAiItems(items);
    72→      setStep(2);
    73→    } catch (e) {
    74→      setError(e.message || "حدث خطأ في المعالجة");
    75→      setStep(0);
    76→    }
    77→  };
    78→
    79→  // ── Merge duplicates by normalizedName ───────────────────────────
    80→  const getMerged = () => {
    81→    const map = new Map();
    82→    for (const it of aiItems) {
    83→      const key = it.normalizedName?.trim().toLowerCase() || it.rawText;
    84→      if (map.has(key)) {
    85→        const ex = map.get(key);
    86→        map.set(key, { ...ex, quantity: ex.quantity + it.quantity });
    87→      } else {
    88→        map.set(key, { ...it });
    89→      }
    90→    }
    91→    return Array.from(map.values());
    92→  };
    93→
    94→  // ── Save Purchase ─────────────────────────────────────────────────
    95→  const savePurchase = async () => {
    96→    setSaving(true);
    97→    try {
    98→      const merged = getMerged();
    99→      const num = `PUR-AI-${Date.now().toString().slice(-6)}`;
   100→      const totalQty = merged.reduce((s, it) => s + it.quantity, 0);
   101→      const items = merged.map(it => ({
   102→        name: it.normalizedName || it.rawText,
   103→        qty: it.quantity,
   104→        purchasePrice: it.purchasePrice ? String(it.purchasePrice) : "",
   105→      }));
   106→      const created = await api.createPurchaseInvoice({
   107→        num,
   108→        date: purchaseForm.date,
   109→        supplier: purchaseForm.supplier,
   110→        companySlug: company?.sk,
   111→        items,
   112→        sourceInvoiceIds: [],
   113→        totalQty,
   114→        notes: purchaseForm.notes || `AI processed: ${rawText.slice(0, 80)}`,
   115→      });
   116→      setStep(4);
   117→      if (onPurchaseSaved) onPurchaseSaved(created);
   118→    } catch (e) {
   119→      toast_("❌ " + (e.message || "حدث خطأ أثناء الحفظ"));
   120→    } finally { setSaving(false); }
   121→  };
   122→
   123→  // ── Catalog CRUD ─────────────────────────────────────────────────
   124→  const addProduct = async () => {
   125→    if (!newProduct.name.trim()) return;
   126→    setAddingProduct(true);
   127→    try {
   128→      const aliases = newProduct.aliases.split(",").map(s => s.trim()).filter(Boolean);
   129→      const created = await api.createCatalogProduct({
   130→        name: newProduct.name.trim(),
   131→        aliases,
   132→        purchasePrice: newProduct.purchasePrice ? pN(newProduct.purchasePrice) : null,
   133→        sellingPrice: newProduct.sellingPrice ? pN(newProduct.sellingPrice) : null,
   134→        companySlug: company?.sk,
   135→      });
   136→      setCatalog(c => [...c, created]);
   137→      setNewProduct({ name: "", aliases: "", purchasePrice: "", sellingPrice: "" });
   138→      toast_("✅ تم إضافة المنتج للكتالوج");
   139→    } catch (e) { toast_("❌ " + e.message); }
   140→    finally { setAddingProduct(false); }
   141→  };
   142→
   143→  const deleteProduct = async (id) => {
   144→    try {
   145→      await api.deleteCatalogProduct(id);
   146→      setCatalog(c => c.filter(x => x.id !== id));
   147→      toast_("🗑️ تم الحذف");
   148→    } catch { toast_("❌ فشل الحذف"); }
   149→  };
   150→
   151→  const updateCatalogItem = async (id, patch) => {
   152→    try {
   153→      const updated = await api.updateCatalogProduct(id, patch);
   154→      setCatalog(c => c.map(x => x.id === id ? { ...x, ...updated } : x));
   155→      setEditingCatalogId(null);
   156→      toast_("✅ تم التحديث");
   157→    } catch { toast_("❌ فشل التحديث"); }
   158→  };
   159→
   160→  // ── Edit AI item ──────────────────────────────────────────────────
   161→  const editItem = (id, field, val) => {
   162→    setAiItems(prev => prev.map(it =>
   163→      it._id === id ? { ...it, [field]: val, edited: true } : it
   164→    ));
   165→  };
   166→
   167→  const removeItem = id => setAiItems(prev => prev.filter(it => it._id !== id));
   168→
   169→  // ── Styles ────────────────────────────────────────────────────────
   170→  const s = {
   171→    card: { background: "#fff", borderRadius: "12px", border: "1px solid #e5e7eb", boxShadow: "0 1px 4px rgba(0,0,0,.07)" },
   172→    btn: (bg, c = "#fff") => ({ border: "none", borderRadius: "8px", padding: "9px 16px", fontFamily: "inherit", fontSize: "13px", fontWeight: 700, cursor: "pointer", background: bg, color: c, display: "inline-flex", alignItems: "center", gap: "5px" }),
   173→    inp: { width: "100%", border: "1.5px solid #d1d5db", borderRadius: "8px", padding: "8px 11px", fontFamily: "inherit", fontSize: "13px", outline: "none" },
   174→    lbl: { fontSize: "11px", color: "#6b7280", fontWeight: 700, display: "block", marginBottom: "4px" },
   175→    sec: { fontSize: "11px", fontWeight: 900, color: "#6b7280", textTransform: "uppercase", letterSpacing: ".5px", marginBottom: "10px" },
   176→  };
   177→
   178→  const merged = step === 2 ? getMerged() : [];
   179→
   180→  return (
   181→    <div style={{ fontFamily: "'Cairo','Tajawal',sans-serif", direction: "rtl" }}>
   182→
   183→      {/* Toast */}
   184→      {toast && (
   185→        <div style={{ position: "fixed", top: 64, left: "50%", transform: "translateX(-50%)", zIndex: 9999, background: toast.startsWith("❌") ? "#dc2626" : "#16a34a", color: "#fff", padding: "8px 22px", borderRadius: "50px", fontWeight: 700, fontSize: "13px", whiteSpace: "nowrap", boxShadow: "0 4px 16px rgba(0,0,0,.2)" }}>{toast}</div>
   186→      )}
   187→
   188→      {/* ── Header ── */}
   189→      <div style={{ display: "flex", alignItems: "center", gap: "10px", marginBottom: "16px", flexWrap: "wrap" }}>
   190→        <div style={{ fontSize: "17px", fontWeight: 900, color: col }}>🤖 معالجة AI للمشتريات</div>
   191→        <div style={{ flex: 1 }} />
   192→        <button style={s.btn("#7c3aed")} onClick={() => setStep(step === 3 ? 0 : 3)}>
   193→          📦 {step === 3 ? "رجوع" : `كتالوج المنتجات (${catalog.length})`}
   194→        </button>
   195→      </div>
   196→
   197→      {/* ══ STEP 0: Input ══════════════════════════════════════════════ */}
   198→      {step === 0 && (
   199→        <div style={{ ...s.card, padding: "20px" }}>
   200→          <div style={s.sec}>أدخل قائمة المنتجات (عربي أو إنجليزي أو مختلط)</div>
   201→          <div style={{ background: "#f8fafc", border: "1px solid #e5e7eb", borderRadius: "8px", padding: "11px 14px", marginBottom: "12px", fontSize: "12px", color: "#6b7280", lineHeight: "1.7" }}>
   202→            <b style={{ color: "#374151" }}>أمثلة:</b> <span dir="ltr">3 m19 headphones, ٥ شواحن ايفون, 2 powerbank 20k, اربع سماعات JBL, cable type c × 10</span>
   203→          </div>
   204→          <textarea
   205→            ref={textRef}
   206→            style={{ ...s.inp, minHeight: "160px", resize: "vertical", marginBottom: "12px", fontSize: "13px", lineHeight: "1.8" }}
   207→            placeholder={"3 m19 headphones\n٥ شواحن ايفون\n2 powerbank 20k\nاربع سماعات JBL\ncable type c × 10"}
   208→            value={rawText}
   209→            onChange={e => setRawText(e.target.value)}
   210→          />
   211→          {error && (
   212→            <div style={{ background: "#fee2e2", border: "1px solid #fca5a5", borderRadius: "8px", padding: "10px 14px", color: "#dc2626", fontSize: "13px", marginBottom: "12px" }}>
   213→              ❌ {error}
   214→            </div>
   215→          )}
   216→          <div style={{ display: "flex", gap: "8px", flexWrap: "wrap" }}>
   217→            <button style={{ ...s.btn(col), flex: 1, justifyContent: "center", padding: "12px" }}
   218→              onClick={processWithAI} disabled={!rawText.trim()}>
   219→              🤖 معالجة بالذكاء الاصطناعي
   220→            </button>
   221→            {catalog.length === 0 && (
   222→              <button style={s.btn("#7c3aed")} onClick={() => setStep(3)}>
   223→                📦 أضف كتالوج المنتجات أولاً
   224→              </button>
   225→            )}
   226→          </div>
   227→          {catalog.length > 0 && (
   228→            <div style={{ marginTop: "10px", fontSize: "12px", color: "#6b7280" }}>
   229→              🔍 سيتم المطابقة مع <b>{catalog.length}</b> منتج في الكتالوج
   230→            </div>
   231→          )}
   232→        </div>
   233→      )}
   234→
   235→      {/* ══ STEP 1: Processing ════════════════════════════════════════ */}
   236→      {step === 1 && (
   237→        <div style={{ ...s.card, padding: "56px", textAlign: "center" }}>
   238→          <div style={{ fontSize: "44px", marginBottom: "14px", animation: "spin 1s linear infinite" }}>⚙️</div>
   239→          <div style={{ fontSize: "16px", fontWeight: 700, color: col, marginBottom: "6px" }}>جارٍ المعالجة بالذكاء الاصطناعي...</div>
   240→          <div style={{ color: "#6b7280", fontSize: "13px" }}>يتم تحليل النص واستخراج المنتجات وتطبيعها</div>
   241→          <style>{`@keyframes spin{from{transform:rotate(0)}to{transform:rotate(360deg)}}`}</style>
   242→        </div>
   243→      )}
   244→
   245→      {/* ══ STEP 2: Review & Edit ════════════════════════════════════ */}
   246→      {step === 2 && (
   247→        <div style={{ display: "flex", flexDirection: "column", gap: "12px" }}>
   248→
   249→          {/* Summary */}
   250→          <div style={{ ...s.card, padding: "14px 18px", display: "flex", gap: "16px", flexWrap: "wrap", alignItems: "center" }}>
   251→            <div>
   252→              <span style={{ fontSize: "12px", color: "#6b7280" }}>المنتجات المكتشفة</span>
   253→              <div style={{ fontWeight: 900, fontSize: "22px", color: col }}>{aiItems.length}</div>
   254→            </div>
   255→            <div>
   256→              <span style={{ fontSize: "12px", color: "#6b7280" }}>بعد الدمج</span>
   257→              <div style={{ fontWeight: 900, fontSize: "22px", color: "#7c3aed" }}>{merged.length}</div>
   258→            </div>
   259→            <div>
   260→              <span style={{ fontSize: "12px", color: "#6b7280" }}>إجمالي الكميات</span>
   261→              <div style={{ fontWeight: 900, fontSize: "22px", color: "#16a34a" }}>{merged.reduce((s, it) => s + it.quantity, 0)}</div>
   262→            </div>
   263→            <div style={{ flex: 1 }} />
   264→            <button style={s.btn("#e5e7eb", "#374151")} onClick={() => { setStep(0); setAiItems([]); }}>← إعادة الإدخال</button>
   265→          </div>
   266→
   267→          {/* Individual AI Results */}
   268→          <div style={{ ...s.card, overflow: "hidden" }}>
   269→            <div style={{ padding: "14px 18px 8px", borderBottom: "1px solid #f0f0f0" }}>
   270→              <div style={s.sec}>نتائج الذكاء الاصطناعي — راجع وعدّل قبل الحفظ</div>
   271→            </div>
   272→            <div style={{ overflowX: "auto" }}>
   273→              <table style={{ width: "100%", borderCollapse: "collapse", fontSize: "13px" }}>
   274→                <thead>
   275→                  <tr style={{ background: "#f8fafc" }}>
   276→                    <th style={{ padding: "9px 12px", textAlign: "right", color: "#6b7280", fontWeight: 700, fontSize: "11px" }}>النص الأصلي</th>
   277→                    <th style={{ padding: "9px 12px", textAlign: "right", color: "#6b7280", fontWeight: 700, fontSize: "11px" }}>الاسم المطبّع</th>
   278→                    <th style={{ padding: "9px 12px", textAlign: "center", color: "#6b7280", fontWeight: 700, fontSize: "11px", width: "80px" }}>الكمية</th>
   279→                    <th style={{ padding: "9px 12px", textAlign: "right", color: "#6b7280", fontWeight: 700, fontSize: "11px", width: "120px" }}>سعر الشراء KD</th>
   280→                    <th style={{ padding: "9px 12px", textAlign: "center", color: "#6b7280", fontWeight: 700, fontSize: "11px", width: "90px" }}>الثقة</th>
   281→                    <th style={{ padding: "9px 12px", width: "40px" }}></th>
   282→                  </tr>
   283→                </thead>
   284→                <tbody>
   285→                  {aiItems.map((it) => (
   286→                    <tr key={it._id} style={{ borderBottom: "1px solid #f5f5f5", background: it.edited ? "#fefce8" : "#fff" }}>
   287→                      <td style={{ padding: "8px 12px", color: "#9ca3af", fontSize: "12px", maxWidth: "140px" }}>
   288→                        <div style={{ overflow: "hidden", textOverflow: "ellipsis", whiteSpace: "nowrap" }}>{it.rawText}</div>
   289→                        {it.catalogName && (
   290→                          <div style={{ color: "#7c3aed", fontSize: "11px", marginTop: "2px" }}>🔗 {it.catalogName}</div>
   291→                        )}
   292→                      </td>
   293→                      <td style={{ padding: "6px 12px" }}>
   294→                        <input
   295→                          style={{ ...s.inp, padding: "5px 8px", fontWeight: 700 }}
   296→                          value={it.normalizedName}
   297→                          onChange={e => editItem(it._id, "normalizedName", e.target.value)}
   298→                        />
   299→                      </td>
   300→                      <td style={{ padding: "6px 12px", textAlign: "center" }}>
   301→                        <input
   302→                          style={{ ...s.inp, padding: "5px 8px", textAlign: "center", fontWeight: 900, width: "64px" }}
   303→                          type="number" min="1" value={it.quantity}
   304→                          onChange={e => editItem(it._id, "quantity", Number(e.target.value) || 1)}
   305→                        />
   306→                      </td>
   307→                      <td style={{ padding: "6px 12px" }}>
   308→                        <input
   309→                          style={{ ...s.inp, padding: "5px 8px", width: "100px" }}
   310→                          type="number" step="0.001" min="0" placeholder="0.000"
   311→                          value={it.purchasePrice ?? ""}
   312→                          onChange={e => editItem(it._id, "purchasePrice", e.target.value)}
   313→                        />
   314→                      </td>
   315→                      <td style={{ padding: "8px 12px", textAlign: "center" }}>
   316→                        <div style={{ display: "flex", flexDirection: "column", alignItems: "center", gap: "2px" }}>
   317→                          <span style={{ background: CONF_BG(it.confidence), color: CONF_COLOR(it.confidence), borderRadius: "20px", padding: "2px 8px", fontSize: "11px", fontWeight: 700 }}>
   318→                            {CONF_LABEL(it.confidence)}
   319→                          </span>
   320→                          <span style={{ color: "#9ca3af", fontSize: "10px" }}>{Math.round(it.confidence * 100)}%</span>
   321→                        </div>
   322→                      </td>
   323→                      <td style={{ padding: "6px 8px", textAlign: "center" }}>
   324→                        <button onClick={() => removeItem(it._id)}
   325→                          style={{ background: "none", border: "none", cursor: "pointer", color: "#dc2626", fontSize: "16px", padding: "2px" }}>✕</button>
   326→                      </td>
   327→                    </tr>
   328→                  ))}
   329→                </tbody>
   330→              </table>
   331→            </div>
   332→          </div>
   333→
   334→          {/* Merged Preview */}
   335→          {merged.length > 0 && (
   336→            <div style={{ ...s.card, overflow: "hidden" }}>
   337→              <div style={{ padding: "14px 18px 8px", borderBottom: "1px solid #f0f0f0" }}>
   338→                <div style={s.sec}>📦 المنتجات المدمجة — هذا ما سيُحفظ في فاتورة المشتريات</div>
   339→              </div>
   340→              <table style={{ width: "100%", borderCollapse: "collapse", fontSize: "13px" }}>
   341→                <thead>
   342→                  <tr style={{ background: col + "15" }}>
   343→                    <th style={{ padding: "8px 14px", textAlign: "right", color: col, fontWeight: 700 }}>المنتج</th>
   344→                    <th style={{ padding: "8px 14px", textAlign: "center", color: col, fontWeight: 700 }}>الكمية الكلية</th>
   345→                    <th style={{ padding: "8px 14px", textAlign: "right", color: col, fontWeight: 700 }}>سعر الشراء</th>
   346→                    <th style={{ padding: "8px 14px", textAlign: "right", color: col, fontWeight: 700 }}>الإجمالي</th>
   347→                  </tr>
   348→                </thead>
   349→                <tbody>
   350→                  {merged.map((it, i) => {
   351→                    const price = pN(it.purchasePrice);
   352→                    const total = price * it.quantity;
   353→                    return (
   354→                      <tr key={i} style={{ borderBottom: "1px solid #f5f5f5", background: i % 2 === 0 ? "#fff" : "#fafafa" }}>
   355→                        <td style={{ padding: "9px 14px", fontWeight: 700 }}>{it.normalizedName}</td>
   356→                        <td style={{ padding: "9px 14px", textAlign: "center", fontWeight: 900, fontSize: "16px", color: col }}>{it.quantity}</td>
   357→                        <td style={{ padding: "9px 14px", color: price ? "#dc2626" : "#9ca3af" }}>{price ? `${price.toFixed(3)} KD` : "—"}</td>
   358→                        <td style={{ padding: "9px 14px", fontWeight: 700 }}>{total ? `${total.toFixed(3)} KD` : "—"}</td>
   359→                      </tr>
   360→                    );
   361→                  })}
   362→                </tbody>
   363→                {merged.reduce((s, it) => s + pN(it.purchasePrice) * it.quantity, 0) > 0 && (
   364→                  <tfoot>
   365→                    <tr style={{ background: col + "20" }}>
   366→                      <td colSpan={3} style={{ padding: "10px 14px", fontWeight: 900, color: col }}>الإجمالي الكلي</td>
   367→                      <td style={{ padding: "10px 14px", fontWeight: 900, color: "#dc2626", fontSize: "14px" }}>
   368→                        {merged.reduce((s, it) => s + pN(it.purchasePrice) * it.quantity, 0).toFixed(3)} KD
   369→                      </td>
   370→                    </tr>
   371→                  </tfoot>
   372→                )}
   373→              </table>
   374→            </div>
   375→          )}
   376→
   377→          {/* Purchase form */}
   378→          <div style={{ ...s.card, padding: "16px 18px" }}>
   379→            <div style={s.sec}>بيانات فاتورة المشتريات</div>
   380→            <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: "12px", marginBottom: "12px" }}>
   381→              <div>
   382→                <label style={s.lbl}>اسم المورد</label>
   383→                <input style={s.inp} placeholder="اختياري" value={purchaseForm.supplier}
   384→                  onChange={e => setPurchaseForm(f => ({ ...f, supplier: e.target.value }))} />
   385→              </div>
   386→              <div>

   506→function CatalogRow({ prod, i, s, col, onDelete, onUpdate, editing, onEdit, onCancelEdit }) {
   507→  const [form, setForm] = useState({
   508→    name: prod.name,
   509→    aliases: Array.isArray(prod.aliases) ? prod.aliases.join(", ") : "",
   510→    purchasePrice: prod.purchasePrice ?? "",
   511→    sellingPrice: prod.sellingPrice ?? "",
   512→  });
   513→
   514→  const save = () => {
   515→    const aliases = form.aliases.split(",").map(x => x.trim()).filter(Boolean);
   516→    onUpdate(prod.id, {
   517→      name: form.name.trim(),
   518→      aliases,
   519→      purchasePrice: form.purchasePrice ? parseFloat(form.purchasePrice) : null,
   520→      sellingPrice: form.sellingPrice ? parseFloat(form.sellingPrice) : null,
   521→    });
   522→  };
   523→
   524→  if (editing) {
   525→    return (
   526→      <tr style={{ borderBottom: "1px solid #e5e7eb", background: "#fefce8" }}>
   527→        <td style={{ padding: "6px 10px" }}><input style={{ ...s.inp, padding: "5px 8px" }} value={form.name} onChange={e => setForm(f => ({ ...f, name: e.target.value }))} /></td>
   528→        <td style={{ padding: "6px 10px" }}><input style={{ ...s.inp, padding: "5px 8px", fontSize: "12px" }} placeholder="alias1, alias2" value={form.aliases} onChange={e => setForm(f => ({ ...f, aliases: e.target.value }))} /></td>
   529→        <td style={{ padding: "6px 10px" }}><input style={{ ...s.inp, padding: "5px 8px", width: "90px" }} type="number" step="0.001" value={form.purchasePrice} onChange={e => setForm(f => ({ ...f, purchasePrice: e.target.value }))} /></td>
   530→        <td style={{ padding: "6px 10px" }}><input style={{ ...s.inp, padding: "5px 8px", width: "90px" }} type="number" step="0.001" value={form.sellingPrice} onChange={e => setForm(f => ({ ...f, sellingPrice: e.target.value }))} /></td>
   531→        <td style={{ padding: "6px 8px" }}>
   532→          <div style={{ display: "flex", gap: "4px" }}>
   533→            <button onClick={save} style={{ background: col, color: "#fff", border: "none", borderRadius: "6px", padding: "4px 10px", cursor: "pointer", fontSize: "12px", fontWeight: 700 }}>✓</button>
   534→            <button onClick={onCancelEdit} style={{ background: "#e5e7eb", border: "none", borderRadius: "6px", padding: "4px 8px", cursor: "pointer", fontSize: "12px" }}>✕</button>
   535→          </div>
   536→        </td>
   537→      </tr>
   538→    );
   539→  }
   540→
   541→  return (
   542→    <tr style={{ borderBottom: "1px solid #f5f5f5", background: i % 2 === 0 ? "#fff" : "#fafafa" }}>
   543→      <td style={{ padding: "9px 14px", fontWeight: 700 }}>{prod.name}</td>
   544→      <td style={{ padding: "9px 14px", fontSize: "12px", color: "#6b7280" }}>
   545→        {Array.isArray(prod.aliases) && prod.aliases.length > 0 ? prod.aliases.join("، ") : "—"}
   546→      </td>
   547→      <td style={{ padding: "9px 14px", color: prod.purchasePrice ? "#dc2626" : "#9ca3af", fontWeight: prod.purchasePrice ? 700 : 400 }}>
   548→        {prod.purchasePrice ? `${parseFloat(prod.purchasePrice).toFixed(3)} KD` : "—"}
   549→      </td>
   550→      <td style={{ padding: "9px 14px", color: prod.sellingPrice ? "#16a34a" : "#9ca3af", fontWeight: prod.sellingPrice ? 700 : 400 }}>
   551→        {prod.sellingPrice ? `${parseFloat(prod.sellingPrice).toFixed(3)} KD` : "—"}
   552→      </td>
   553→      <td style={{ padding: "9px 8px" }}>
   554→        <div style={{ display: "flex", gap: "4px" }}>
   555→          <button onClick={onEdit} style={{ background: "#fef3c7", color: "#92400e", border: "none", borderRadius: "6px", padding: "4px 8px", cursor: "pointer", fontSize: "12px" }}>✏️</button>
   556→          <button onClick={() => onDelete(prod.id)} style={{ background: "#fee2e2", color: "#dc2626", border: "none", borderRadius: "6px", padding: "4px 8px", cursor: "pointer", fontSize: "12px" }}>🗑️</button>
   557→        </div>
   558→      </td>
   559→    </tr>
   560→  );
   561→}
